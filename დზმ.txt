// მუსიკის ღილაკი
let audio = gio pika-gdze ti;
let timer = null;
let note = 0;

const notes = [523, 659, 784, 659, 587, 698, 880, 698];

function playNote() {
  const osc = audio.createOscillator();
  const gain = audio.createGain();

  osc.type = "sine";
  osc.frequency.value = notes[note++ % notes.length];

  gain.gain.setValueAtTime(0.06, audio.currentTime);
  gain.gain.exponentialRampToValueAtTime(
    0.001,
    audio.currentTime + 0.35
  );

  osc.connect(gain);
  gain.connect(audio.destination);

  osc.start();
  osc.stop(audio.currentTime + 0.36);
}

document.getElementById("music").onclick = async function() {
  if (timer) {
    clearInterval(timer);
    timer = null;

    await audio.suspend();

    this.textContent = "♫ მუსიკის ჩართვა ქურდულად";
    return;
  }

  audio = audio || new AudioContext();

  await audio.resume();

  playNote();

  timer = setInterval(playNote, 450);

  this.textContent = "♫ მუსიკის ზადნის მიცემა";
};
